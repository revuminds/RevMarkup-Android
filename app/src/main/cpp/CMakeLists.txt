cmake_minimum_required(VERSION 3.22)
project(revmarkup C)
set(PDFIUM ${CMAKE_CURRENT_SOURCE_DIR}/pdfium)
add_library(pdfium SHARED IMPORTED)
set_target_properties(pdfium PROPERTIES IMPORTED_LOCATION ${PDFIUM}/lib/libpdfium.so)
add_library(revpdf SHARED pdfium_jni.c)
target_include_directories(revpdf PRIVATE ${PDFIUM}/include)
target_compile_options(revpdf PRIVATE -O2 -Wall -fstack-protector-strong -D_FORTIFY_SOURCE=2)
target_link_options(revpdf PRIVATE -Wl,-z,relro,-z,now)
target_link_libraries(revpdf pdfium jnigraphics log)
